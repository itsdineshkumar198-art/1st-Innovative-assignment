STEP 1: Create a Product class with private attributes productId, productName, and price.

STEP 2: Use a constructor to initialize the product details.

STEP 3: Create getter methods to access private variables.

STEP 4: Create a Customer class with customerId and customerName.

STEP 5: Initialize customer details using a constructor and provide getter methods.

java code:

// Product Class
class Product {
    private int productId;
    private String productName;
    private double price;

    // Constructor
    Product(int productId, String productName, double price) {
        this.productId = productId;
        this.productName = productName;
        this.price = price;
    }

    // Getter Methods
    public int getProductId() {
        return productId;
    }

    public String getProductName() {
        return productName;
    }

    public double getPrice() {
        return price;
    }
}

// Customer Class
class Customer {
    private int customerId;
    private String customerName;

    // Constructor
    Customer(int customerId, String customerName) {
        this.customerId = customerId;
        this.customerName = customerName;
    }

    // Getter Methods
    public int getCustomerId() {
        return customerId;
    }

    public String getCustomerName() {
        return customerName;
    }
}

// Order Class
class Order {
    private Customer customer;
    private Product[] products;
    private int[] quantity;

    // Constructor
    Order(Customer customer, Product[] products, int[] quantity) {
        this.customer = customer;
        this.products = products;
        this.quantity = quantity;
    }

    // Display Order Details
    void displayOrder() {

        System.out.println("----- E-Commerce Order Details -----");

        // String Processing
        System.out.println("Customer Name : " +
                customer.getCustomerName().toUpperCase());

        System.out.println("Customer ID   : " +
                customer.getCustomerId());

        double total = 0;

        System.out.println("\nProducts Ordered:");

        for (int i = 0; i < products.length; i++) {

            double amount = products[i].getPrice() * quantity[i];

            total += amount;

            System.out.println(products[i].getProductName()
                    + " | Price: Rs." + products[i].getPrice()
                    + " | Quantity: " + quantity[i]
                    + " | Amount: Rs." + amount);
        }

        System.out.println("\nTotal Bill : Rs." + total);
    }
}

// Main Class
public class EcommerceManagement {

    public static void main(String[] args) {

        // Product Objects
        Product p1 = new Product(101, "Laptop", 55000);
        Product p2 = new Product(102, "Wireless Mouse", 1200);
        Product p3 = new Product(103, "Keyboard", 1800);

        // Customer Object
        Customer c1 = new Customer(1, "Dinesh Kumar");

        // Arrays
        Product[] productList = {p1, p2, p3};
        int[] quantity = {1, 2, 1};

        // Order Object
        Order order = new Order(c1, productList, quantity);

        // Display Order
        order.displayOrder();
    }
}
